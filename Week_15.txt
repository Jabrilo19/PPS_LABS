import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.AfterEach;
import static org.junit.jupiter.api.Assertions.*;

// 1b make and run test
class BankAccountTest {

    private BankAccount account;

    @BeforeEach
    void setUp() {
        // Starts each test with a fresh account of 100.0
        account = new BankAccount(100.0);
    }

    // 1. Add an @AfterEach annotation and method to delete the current bank account to make it available for garbage collection
    @AfterEach
    void tearDown() {
        account = null;
    }

    @Test
    void testDeposit() {
        // 2. deposit $50 and check that the balance is 150
        account.deposit(50.0);
        assertEquals(150.0, account.getBalance(), "Balance should be 150.0 after a $50 deposit");
    }

    @Test
    void testWithdraw() {
        // 3. withdraw $40 and check that the balance is $60; remember that each test is done on a fresh instance of bank account
        account.withdraw(40.0);
        assertEquals(60.0, account.getBalance(), "Balance should be 60.0 after a $40 withdrawal");
    }

    @Test
    void testInvalidDeposit() {
        // 4. Deposit a negative amount and check if an exception is thrown
        assertThrows(IllegalArgumentException.class, () -> {
            account.deposit(-20.0);
        }, "Depositing a negative amount should throw an IllegalArgumentException");
    }

    @Test
    void testOverdraft() {
        // 5. Verify that Withdrawing more than the current balance throws an exception
        assertThrows(IllegalArgumentException.class, () -> {
            account.withdraw(200.0);
        }, "Withdrawing more than the balance should throw an IllegalArgumentException");
    }

    // 6. Add a test to check that an Exception is thrown when trying to create a new bankaccout with a negaive initial balance
    @Test
    void testNegativeInitialBalance() {
        assertThrows(IllegalArgumentException.class, () -> {
            new BankAccount(-50.0);
        }, "Creating an account with a negative initial balance should throw an IllegalArgumentException");
    }

    // Test for 1c: Transfer method from one account to another
    @Test
    void testTransfer() {
        BankAccount targetAccount = new BankAccount(50.0);
        account.transfer(targetAccount, 25.0);
        
        assertEquals(75.0, account.getBalance(), "Source account balance should decrease by $25");
        assertEquals(75.0, targetAccount.getBalance(), "Target account balance should increase by $25");
    }
}

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class ComputationsTest {

    @Test
    void testFibonacci() {
        assertEquals(0, Computations.fibonacci(0), "Fibonacci of 0 should be 0");
        assertEquals(1, Computations.fibonacci(1), "Fibonacci of 1 should be 1");
        assertEquals(5, Computations.fibonacci(5), "Fibonacci of 5 should be 5");
        assertEquals(8, Computations.fibonacci(6), "Fibonacci of 6 should be 8");
        
        // Edge case: Negative input should throw exception
        assertThrows(IllegalArgumentException.class, () -> {
            Computations.fibonacci(-1);
        }, "Negative input should throw IllegalArgumentException");
    }

    @Test
    void testIsPrime() {
        assertFalse(Computations.isPrime(-5), "Negative numbers are not prime");
        assertFalse(Computations.isPrime(0), "0 is not prime");
        assertFalse(Computations.isPrime(1), "1 is not prime");
        assertTrue(Computations.isPrime(2), "2 is a prime number (edge case)");
        assertTrue(Computations.isPrime(7), "7 is a prime number");
        assertFalse(Computations.isPrime(9), "9 is not prime (divisible by 3)");
    }

    @Test
    void testIsEven() {
        assertTrue(Computations.isEven(0), "0 is even");
        assertTrue(Computations.isEven(4), "4 is even");
        assertTrue(Computations.isEven(-2), "-2 is even");
        assertFalse(Computations.isEven(3), "3 is not even");
    }

    @Test
    void testIsOdd() {
        assertFalse(Computations.isOdd(0), "0 is not odd");
        assertTrue(Computations.isOdd(3), "3 is odd");
        assertTrue(Computations.isOdd(-5), "-5 is odd");
        assertFalse(Computations.isOdd(4), "4 is not odd");
    }

    @Test
    void testToCelsius() {
        // Using a small delta (0.001) for floating-point comparisons
        assertEquals(0.0, Computations.toCelsius(32.0), 0.001, "32F should be 0C");
        assertEquals(100.0, Computations.toCelsius(212.0), 0.001, "212F should be 100C");
        assertEquals(-40.0, Computations.toCelsius(-40.0), 0.001, "-40F should be -40C");
    }

    @Test
    void testToFahrenheit() {
        assertEquals(32.0, Computations.toFahrenheit(0.0), 0.001, "0C should be 32F");
        assertEquals(212.0, Computations.toFahrenheit(100.0), 0.001, "100C should be 212F");
        assertEquals(-40.0, Computations.toFahrenheit(-40.0), 0.001, "-40C should be -40F");
    }
}
